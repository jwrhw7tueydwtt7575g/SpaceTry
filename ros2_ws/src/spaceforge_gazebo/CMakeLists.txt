cmake_minimum_required(VERSION 3.16)
project(spaceforge_gazebo)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(gz-math7 REQUIRED)
add_library(spaceforge_cw_system SHARED src/cw_dynamics_system.cpp)
add_executable(spaceforge_sim_bridge src/sim_bridge.cpp)
ament_target_dependencies(spaceforge_sim_bridge rclcpp rosgraph_msgs nav_msgs sensor_msgs geometry_msgs)
target_link_libraries(spaceforge_cw_system PRIVATE gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 gz-math7::gz-math7)
# ament_target_dependencies uses the plain target_link_libraries signature, and
# CMake forbids mixing plain and keyword signatures on one target.
target_link_libraries(spaceforge_sim_bridge gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
# Gazebo loads the system plugin from lib/ (GZ_SIM_SYSTEM_PLUGIN_PATH); ROS 2
# launch/run look for package executables in lib/${PROJECT_NAME}.
install(TARGETS spaceforge_cw_system LIBRARY DESTINATION lib)
install(TARGETS spaceforge_sim_bridge RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY worlds models launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md INTERFACES.md SIM_BRIDGE.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(package_layout test/test_package_layout.py)
  ament_add_pytest_test(sim_bridge_contract test/test_sim_bridge_contract.py)
  ament_add_pytest_test(sim_bridge_integration test/test_sim_bridge_integration.py TIMEOUT 120)
endif()
ament_package()
